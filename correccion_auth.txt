import { Request, Response, NextFunction } from "express";

export type NeonUser = {
  id: string;
  email: string;
  name?: string;
};

declare global {
  namespace Express {
    interface Request {
      neonUser?: NeonUser;
    }
  }
}

export async function requireNeonAuth(
  req: Request,
  res: Response,
  next: NextFunction
) {
  try {
    const response = await fetch(
      `${process.env.NEON_AUTH_URL}/session`,
      {
        headers: {
          cookie: req.headers.cookie ?? "",
        },
      }
    );

    if (!response.ok) {
      return res.status(401).json({ error: "Unauthorized" });
    }

    const data = await response.json();

    req.neonUser = {
      id: data.user.id,
      email: data.user.email,
      name: data.user.name,
    };

    next();
  } catch (err) {
    console.error("Neon auth error:", err);
    res.status(401).json({ error: "Unauthorized" });
  }
}

🔍 Qué hace esto

Usa la cookie que setea Neon Auth

Consulta /session del auth server

Inyecta req.neonUser

No expone tokens al frontend

100% seguro y stateless

✅ Endpoint /api/me
app.get("/api/me", requireNeonAuth, async (req, res) => {
  const neonUser = req.neonUser!;

  const existing = await db
    .select()
    .from(users)
    .where(eq(users.id, neonUser.id))
    .limit(1);

  let user = existing[0];

  if (!user) {
    const inserted = await db
      .insert(users)
      .values({
        id: neonUser.id,
        email: neonUser.email,
        name: neonUser.name,
        role: "user",
      })
      .returning();

    user = inserted[0];
  }

  res.json({
    user,
  });
});

🔑 Detalle CLAVE

Este patrón se llama “just-in-time user provisioning”

No necesitas endpoint /register

Neon Auth manda → backend sincroniza

app.post("/api/logout", async (req, res) => {
  try {
    await fetch(`${process.env.NEON_AUTH_URL}/logout`, {
      method: "POST",
      headers: {
        cookie: req.headers.cookie ?? "",
      },
    });
  } catch (err) {
    console.error("Logout error:", err);
  }

  res.clearCookie("neon_session");
  res.json({ success: true });
});
