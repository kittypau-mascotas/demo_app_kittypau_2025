7️⃣ TELEMETRÍA (CORE IoT)

⚠️ TABLA MÁS CRÍTICA DEL SISTEMA

CREATE TABLE IF NOT EXISTS telemetry (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  pet_id UUID NOT NULL REFERENCES pets(id) ON DELETE CASCADE,
  device_id UUID NOT NULL REFERENCES devices(id) ON DELETE CASCADE,

  temperature REAL,
  humidity REAL,
  weight REAL,
  activity REAL,

  created_at TIMESTAMP DEFAULT now() NOT NULL
);

🔥 ÍNDICES CRÍTICOS (NO OPCIONAL)
CREATE INDEX IF NOT EXISTS telemetry_user_pet_idx
  ON telemetry(user_id, pet_id);

CREATE INDEX IF NOT EXISTS telemetry_device_idx
  ON telemetry(device_id);

CREATE INDEX IF NOT EXISTS telemetry_created_idx
  ON telemetry(created_at DESC);

  📌 Nota profesional

AND created_at BETWEEN $3 AND $4
ORDER BY created_at DESC
